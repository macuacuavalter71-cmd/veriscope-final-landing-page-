//@version=6
indicator("Veriscope Trade Pilot", shorttitle = "Trade Pilot", overlay = true,
     max_lines_count = 20, max_labels_count = 20)

GRP_LANG = "Language / Idioma"
langInput = input.string("Português", "Dashboard Language", options = ["English", "Português"], group = GRP_LANG)

f_t(string en, string pt) =>
    langInput == "Português" ? pt : en

f_pos(string p) =>
    switch p
        "Top Right"    => position.top_right
        "Top Left"     => position.top_left
        "Bottom Right" => position.bottom_right
        "Bottom Left"  => position.bottom_left
        => position.top_right

GRP_ID = "Identidade do Trade"
biasInput    = input.string("Compra (Long)", "Direção", options = ["Compra (Long)", "Venda (Short)"], group = GRP_ID)
strategyTag  = input.string("", "Tag da Estratégia", group = GRP_ID)

GRP_CHECK = "Checklist Pré-Trade"
chk1 = input.bool(false, "1. Zona validada no Veriscope Pro", group = GRP_CHECK)
chk2 = input.bool(false, "2. Score aprovado no Veriscope Intelligence", group = GRP_CHECK)
chk3 = input.bool(false, "3. Risco calculado no Veriscope Risk Engine", group = GRP_CHECK)
chk4 = input.bool(false, "4. Sem notícia de alto impacto no horário", group = GRP_CHECK)
chk5 = input.bool(false, "5. Entrada, Stop e Alvos definidos por escrito", group = GRP_CHECK)
chk6 = input.bool(false, "6. Estado emocional neutro (sem tilt)", group = GRP_CHECK)

GRP_PLAN = "Plano de Trade (definido antes da entrada)"
planEntry = input.float(0.0, "Preço de Entrada Planejado", step = 0.00001, group = GRP_PLAN)
planStop  = input.float(0.0, "Stop Loss Planejado (original)", step = 0.00001, group = GRP_PLAN)
planTP1   = input.float(0.0, "Take Profit 1", step = 0.00001, group = GRP_PLAN)
planTP2   = input.float(0.0, "Take Profit 2 (0 = desativado)", step = 0.00001, group = GRP_PLAN)
planTP3   = input.float(0.0, "Take Profit 3 (0 = desativado)", step = 0.00001, group = GRP_PLAN)

GRP_LIVE = "Execução ao Vivo"
planArmed   = input.bool(false, "🔒 Plano Travado (Armed)", group = GRP_LIVE)
tradeLive   = input.bool(false, "🟢 Operação Está Aberta", group = GRP_LIVE)
tradeClosed = input.bool(false, "⏹ Operação Foi Fechada", group = GRP_LIVE)
liveStop    = input.float(0.0, "Stop Ao Vivo (atualize se ajustar)", step = 0.00001, group = GRP_LIVE)

GRP_TIME = "Tempo & Sessão"
entryTime     = input.time(timestamp("2024-01-01 00:00 +0000"), "Horário da Entrada", group = GRP_TIME)
closeTime     = input.time(timestamp("2024-01-01 00:00 +0000"), "Horário do Fechamento", group = GRP_TIME)
useBlackout   = input.bool(false, "Ativar Janela de Blackout", group = GRP_TIME)
blackoutSess  = input.session("0830-0900", "Janela de Blackout", group = GRP_TIME)
blackoutTZ    = input.string("America/New_York", "Fuso da Janela", group = GRP_TIME)

GRP_DISC = "Disciplina Comportamental"
maxDailyTrades   = input.int(3, "Limite de Trades por Dia", minval = 1, group = GRP_DISC)
tradesTakenToday = input.int(0, "Trades Já Fechados Hoje", minval = 0, group = GRP_DISC)
lastTradeWasLoss = input.bool(false, "Última Operação Foi Perda?", group = GRP_DISC)
cooldownMinutes  = input.int(30, "Cooldown Após Perda (minutos)", minval = 0, group = GRP_DISC)

GRP_VIS = "Dashboard & Visuais"
showDashboard = input.bool(true, "Mostrar Dashboard", group = GRP_VIS)
dashPos       = input.string("Top Right", "Posição do Dashboard", options = ["Top Right", "Top Left", "Bottom Right", "Bottom Left"], group = GRP_VIS)
showPlanLines = input.bool(true, "Mostrar Linhas do Plano no Gráfico", group = GRP_VIS)

// ---------------------------------------------------------------------------
// Cálculos base
// ---------------------------------------------------------------------------
biasLong    = biasInput == "Compra (Long)"
riskPerUnit = math.abs(planEntry - planStop)
hasValidPlan = planEntry != 0 and planStop != 0 and planEntry != planStop

rr1 = (hasValidPlan and planTP1 != 0) ? math.abs(planTP1 - planEntry) / riskPerUnit : na
rr2 = (hasValidPlan and planTP2 != 0) ? math.abs(planTP2 - planEntry) / riskPerUnit : na
rr3 = (hasValidPlan and planTP3 != 0) ? math.abs(planTP3 - planEntry) / riskPerUnit : na

checklistScore    = (chk1 ? 1 : 0) + (chk2 ? 1 : 0) + (chk3 ? 1 : 0) + (chk4 ? 1 : 0) + (chk5 ? 1 : 0) + (chk6 ? 1 : 0)
checklistComplete = checklistScore == 6

armedWithoutChecklist = planArmed and not checklistComplete
liveWithoutArmed       = tradeLive and not planArmed
liveWithoutValidPlan    = tradeLive and not hasValidPlan

status = tradeClosed ? f_t("CLOSED", "FECHADO") : tradeLive ? f_t("IN TRADE", "EM OPERAÇÃO") : planArmed ? f_t("PLANNED", "PLANEJADO") : f_t("IDLE", "OCIOSO")

// ---------------------------------------------------------------------------
// Janela real da operação (usa 'time', que é série — funciona mesmo com inputs constantes)
// ---------------------------------------------------------------------------
inTradeWindow = tradeLive and time >= entryTime and (not tradeClosed or time <= closeTime)

favNow = biasLong ? high : low
advNow = biasLong ? low : high

var float favExtreme = na
var float advExtreme = na

if inTradeWindow
    favExtreme := na(favExtreme) ? favNow : (biasLong ? math.max(favExtreme, favNow) : math.min(favExtreme, favNow))
    advExtreme := na(advExtreme) ? advNow : (biasLong ? math.min(advExtreme, advNow) : math.max(advExtreme, advNow))

if not tradeLive
    favExtreme := na
    advExtreme := na

currentR = hasValidPlan ? (biasLong ? (close - planEntry) : (planEntry - close)) / riskPerUnit : na
mfeR = (hasValidPlan and not na(favExtreme)) ? (biasLong ? (favExtreme - planEntry) : (planEntry - favExtreme)) / riskPerUnit : na
maeR = (hasValidPlan and not na(advExtreme)) ? (biasLong ? (advExtreme - planEntry) : (planEntry - advExtreme)) / riskPerUnit : na

// ---------------------------------------------------------------------------
// Watchdog de desvio de stop
// ---------------------------------------------------------------------------
liveStopSet   = liveStop != 0
riskIncreased = liveStopSet and (biasLong ? liveStop < planStop : liveStop > planStop)
riskReduced   = liveStopSet and liveStop != planStop and not riskIncreased

var bool riskFlagPrev = false
riskJustIncreased = riskIncreased and not riskFlagPrev
riskFlagPrev := riskIncreased

// ---------------------------------------------------------------------------
// Lembretes de parciais (edge-triggered, resetam quando a operação não está aberta)
// ---------------------------------------------------------------------------
tp1Level = planTP1 != 0 and inTradeWindow and (biasLong ? high >= planTP1 : low <= planTP1)
tp2Level = planTP2 != 0 and inTradeWindow and (biasLong ? high >= planTP2 : low <= planTP2)
tp3Level = planTP3 != 0 and inTradeWindow and (biasLong ? high >= planTP3 : low <= planTP3)

var bool tp1Hit = false
var bool tp2Hit = false
var bool tp3Hit = false

tp1JustHit = tp1Level and not tp1Hit
tp2JustHit = tp2Level and not tp2Hit
tp3JustHit = tp3Level and not tp3Hit

tp1Hit := tp1Hit or tp1Level
tp2Hit := tp2Hit or tp2Level
tp3Hit := tp3Hit or tp3Level

if not tradeLive
    tp1Hit := false
    tp2Hit := false
    tp3Hit := false

// ---------------------------------------------------------------------------
// Blackout / cooldown / limite diário
// ---------------------------------------------------------------------------
inBlackout        = useBlackout and not na(time(timeframe.period, blackoutSess, blackoutTZ))
blackoutWhileLive = inBlackout and tradeLive

cooldownActive        = lastTradeWasLoss and time >= closeTime and (time - closeTime) < cooldownMinutes * 60000
newEntryDuringCooldown = tradeLive and cooldownActive
dailyLimitReached      = tradesTakenToday >= maxDailyTrades

// ---------------------------------------------------------------------------
// Linhas do plano no gráfico
// ---------------------------------------------------------------------------
var line entryLine = na
var line stopLine  = na
var line tp1Line   = na
var line tp2Line   = na
var line tp3Line   = na

if showPlanLines and hasValidPlan
    if na(entryLine)
        entryLine := line.new(bar_index, planEntry, bar_index, planEntry, color = color.white, style = line.style_dashed, extend = extend.right)
        stopLine  := line.new(bar_index, planStop, bar_index, planStop, color = color.red, style = line.style_dashed, extend = extend.right)
    else
        line.set_xy1(entryLine, bar_index - 1, planEntry)
        line.set_xy2(entryLine, bar_index, planEntry)
        line.set_xy1(stopLine, bar_index - 1, planStop)
        line.set_xy2(stopLine, bar_index, planStop)

    if planTP1 != 0
        if na(tp1Line)
            tp1Line := line.new(bar_index, planTP1, bar_index, planTP1, color = color.lime, style = line.style_dashed, extend = extend.right)
        else
            line.set_xy1(tp1Line, bar_index - 1, planTP1)
            line.set_xy2(tp1Line, bar_index, planTP1)
    if planTP2 != 0
        if na(tp2Line)
            tp2Line := line.new(bar_index, planTP2, bar_index, planTP2, color = color.green, style = line.style_dashed, extend = extend.right)
        else
            line.set_xy1(tp2Line, bar_index - 1, planTP2)
            line.set_xy2(tp2Line, bar_index, planTP2)
    if planTP3 != 0
        if na(tp3Line)
            tp3Line := line.new(bar_index, planTP3, bar_index, planTP3, color = color.teal, style = line.style_dashed, extend = extend.right)
        else
            line.set_xy1(tp3Line, bar_index - 1, planTP3)
            line.set_xy2(tp3Line, bar_index, planTP3)

if not hasValidPlan
    if not na(entryLine)
        line.delete(entryLine)
        entryLine := na
    if not na(stopLine)
        line.delete(stopLine)
        stopLine := na
if planTP1 == 0 and not na(tp1Line)
    line.delete(tp1Line)
    tp1Line := na
if planTP2 == 0 and not na(tp2Line)
    line.delete(tp2Line)
    tp2Line := na
if planTP3 == 0 and not na(tp3Line)
    line.delete(tp3Line)
    tp3Line := na

// ---------------------------------------------------------------------------
// Tempo em operação
// ---------------------------------------------------------------------------
elapsedMs = tradeClosed ? (closeTime - entryTime) : (tradeLive ? (timenow - entryTime) : na)
elapsedMinTotal = na(elapsedMs) ? na : math.max(elapsedMs, 0) / 60000
elapsedHours = na(elapsedMinTotal) ? na : math.floor(elapsedMinTotal / 60)
elapsedMins  = na(elapsedMinTotal) ? na : math.floor(elapsedMinTotal % 60)

// ---------------------------------------------------------------------------
// Dashboard
// ---------------------------------------------------------------------------
if showDashboard and barstate.islast
    var table dash = table.new(f_pos(dashPos), 2, 25, border_width = 1, bgcolor = color.new(color.black, 15))
    r = 0
    table.cell(dash, 0, r, "VERISCOPE TRADE PILOT", text_color = color.white, bgcolor = color.new(color.black, 0), text_size = size.small)
    table.cell(dash, 1, r, "", bgcolor = color.new(color.black, 0))
    r += 1
    table.cell(dash, 0, r, f_t("Status", "Status"), text_color = color.gray)
    table.cell(dash, 1, r, status, text_color = status == f_t("IN TRADE","EM OPERAÇÃO") ? color.lime : status == f_t("CLOSED","FECHADO") ? color.gray : color.yellow)
    r += 1
    table.cell(dash, 0, r, f_t("Bias / Tag", "Direção / Tag"), text_color = color.gray)
    table.cell(dash, 1, r, biasInput + (strategyTag == "" ? "" : " · " + strategyTag), text_color = color.white)
    r += 1
    table.cell(dash, 0, r, f_t("Checklist", "Checklist"), text_color = color.gray)
    table.cell(dash, 1, r, str.tostring(checklistScore) + "/6", text_color = checklistComplete ? color.lime : color.orange)
    r += 1
    table.cell(dash, 0, r, f_t("PLAN", "PLANO"), text_color = color.white, bgcolor = color.new(color.gray, 70), text_size = size.tiny)
    table.cell(dash, 1, r, "", bgcolor = color.new(color.gray, 70))
    r += 1
    table.cell(dash, 0, r, f_t("Entry / Stop", "Entrada / Stop"), text_color = color.gray, text_size = size.tiny)
    table.cell(dash, 1, r, str.tostring(planEntry, format.mintick) + " / " + str.tostring(planStop, format.mintick), text_color = color.white, text_size = size.tiny)
    r += 1
    table.cell(dash, 0, r, "TP1 (R:R)", text_color = color.gray, text_size = size.tiny)
    table.cell(dash, 1, r, na(rr1) ? "—" : str.tostring(planTP1, format.mintick) + "  (" + str.tostring(rr1, "#.##") + "R)", text_color = color.lime, text_size = size.tiny)
    r += 1
    table.cell(dash, 0, r, "TP2 (R:R)", text_color = color.gray, text_size = size.tiny)
    table.cell(dash, 1, r, na(rr2) ? "—" : str.tostring(planTP2, format.mintick) + "  (" + str.tostring(rr2, "#.##") + "R)", text_color = color.lime, text_size = size.tiny)
    r += 1
    table.cell(dash, 0, r, "TP3 (R:R)", text_color = color.gray, text_size = size.tiny)
    table.cell(dash, 1, r, na(rr3) ? "—" : str.tostring(planTP3, format.mintick) + "  (" + str.tostring(rr3, "#.##") + "R)", text_color = color.lime, text_size = size.tiny)
    r += 1
    table.cell(dash, 0, r, f_t("LIVE", "AO VIVO"), text_color = color.white, bgcolor = color.new(color.gray, 70), text_size = size.tiny)
    table.cell(dash, 1, r, "", bgcolor = color.new(color.gray, 70))
    r += 1
    table.cell(dash, 0, r, f_t("Live Stop", "Stop Ao Vivo"), text_color = color.gray, text_size = size.tiny)
    table.cell(dash, 1, r, liveStopSet ? str.tostring(liveStop, format.mintick) : "—", text_color = riskIncreased ? color.red : riskReduced ? color.lime : color.white, text_size = size.tiny)
    r += 1
    table.cell(dash, 0, r, f_t("Current R", "R Atual"), text_color = color.gray, text_size = size.tiny)
    table.cell(dash, 1, r, na(currentR) ? "—" : str.tostring(currentR, "#.##") + "R", text_color = currentR > 0 ? color.lime : color.red, text_size = size.tiny)
    r += 1
    table.cell(dash, 0, r, "MFE", text_color = color.gray, text_size = size.tiny)
    table.cell(dash, 1, r, na(mfeR) ? "—" : str.tostring(mfeR, "#.##") + "R", text_color = color.lime, text_size = size.tiny)
    r += 1
    table.cell(dash, 0, r, "MAE", text_color = color.gray, text_size = size.tiny)
    table.cell(dash, 1, r, na(maeR) ? "—" : str.tostring(maeR, "#.##") + "R", text_color = color.red, text_size = size.tiny)
    r += 1
    table.cell(dash, 0, r, f_t("Time in Trade", "Tempo em Operação"), text_color = color.gray, text_size = size.tiny)
    table.cell(dash, 1, r, na(elapsedHours) ? "—" : str.tostring(elapsedHours) + "h " + str.tostring(elapsedMins) + "m", text_color = color.white, text_size = size.tiny)
    r += 1
    table.cell(dash, 0, r, f_t("DISCIPLINE", "DISCIPLINA"), text_color = color.white, bgcolor = color.new(color.gray, 70), text_size = size.tiny)
    table.cell(dash, 1, r, "", bgcolor = color.new(color.gray, 70))
    r += 1
    table.cell(dash, 0, r, f_t("Checklist Gate", "Checklist"), text_color = color.gray, text_size = size.tiny)
    table.cell(dash, 1, r, armedWithoutChecklist ? f_t("⚠ INCOMPLETE","⚠ INCOMPLETO") : "OK", text_color = armedWithoutChecklist ? color.red : color.lime, text_size = size.tiny)
    r += 1
    table.cell(dash, 0, r, f_t("Entry Gate", "Entrada"), text_color = color.gray, text_size = size.tiny)
    table.cell(dash, 1, r, liveWithoutArmed ? f_t("⚠ NO PLAN","⚠ SEM PLANO") : "OK", text_color = liveWithoutArmed ? color.red : color.lime, text_size = size.tiny)
    r += 1
    table.cell(dash, 0, r, f_t("Stop Risk", "Risco do Stop"), text_color = color.gray, text_size = size.tiny)
    table.cell(dash, 1, r, riskIncreased ? f_t("⚠ INCREASED","⚠ AUMENTADO") : riskReduced ? f_t("Reduced","Reduzido") : "OK", text_color = riskIncreased ? color.red : riskReduced ? color.lime : color.gray, text_size = size.tiny)
    r += 1
    table.cell(dash, 0, r, f_t("Trades Today", "Trades Hoje"), text_color = color.gray, text_size = size.tiny)
    table.cell(dash, 1, r, str.tostring(tradesTakenToday) + "/" + str.tostring(maxDailyTrades), text_color = dailyLimitReached ? color.orange : color.white, text_size = size.tiny)
    r += 1
    table.cell(dash, 0, r, f_t("Cooldown", "Cooldown"), text_color = color.gray, text_size = size.tiny)
    table.cell(dash, 1, r, cooldownActive ? f_t("ACTIVE","ATIVO") : f_t("Free","Livre"), text_color = cooldownActive ? color.red : color.lime, text_size = size.tiny)
    r += 1
    table.cell(dash, 0, r, f_t("Blackout", "Blackout"), text_color = color.gray, text_size = size.tiny)
    table.cell(dash, 1, r, inBlackout ? f_t("ACTIVE","ATIVO") : f_t("Free","Livre"), text_color = inBlackout ? color.orange : color.lime, text_size = size.tiny)
    r += 1
    table.cell(dash, 0, r, f_t("⚠ Copilot, not autopilot", "⚠ Copiloto, não piloto automático"), text_color = color.orange, text_size = size.tiny)
    table.cell(dash, 1, r, f_t("You execute.", "Você executa."), text_color = color.orange, text_size = size.tiny)

// ---------------------------------------------------------------------------
// Alertas
// ---------------------------------------------------------------------------
alertcondition(armedWithoutChecklist, "Trade Pilot — Checklist Incompleto", "Plano travado mas o checklist não foi concluído. Não execute a entrada.")
alertcondition(liveWithoutArmed, "Trade Pilot — Entrada Sem Plano", "Operação marcada como aberta sem um plano travado. Disciplina comprometida.")
alertcondition(riskJustIncreased, "Trade Pilot — Risco Aumentado", "O stop ao vivo foi movido para AUMENTAR o risco em relação ao plano original.")
alertcondition(tp1JustHit, "Trade Pilot — TP1 Atingido", "Preço atingiu o TP1. Lembrete: realizar parcial conforme o plano.")
alertcondition(tp2JustHit, "Trade Pilot — TP2 Atingido", "Preço atingiu o TP2. Lembrete: realizar parcial conforme o plano.")
alertcondition(tp3JustHit, "Trade Pilot — TP3 Atingido", "Preço atingiu o TP3 (alvo final).")
alertcondition(blackoutWhileLive, "Trade Pilot — Blackout Ativo", "Operação aberta durante uma janela de notícia/evento de alto impacto.")
alertcondition(newEntryDuringCooldown, "Trade Pilot — Entrada Durante Cooldown", "Operação aberta durante o período de cooldown pós-perda.")
alertcondition(dailyLimitReached and tradeLive, "Trade Pilot — Limite Diário Atingido", "Limite de trades definido para hoje já foi atingido.")

plot(currentR, "Current R", display = display.data_window)
plot(mfeR, "MFE (R)", display = display.data_window)
plot(maeR, "MAE (R)", display = display.data_window)

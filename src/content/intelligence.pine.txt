f_dashPosition(string p) =>
    switch p
        "Top Right" => position.top_right
        "Top Left" => position.top_left
        "Bottom Right" => position.bottom_right
        "Bottom Left" => position.bottom_left
        => position.top_right

f_themeColor(bool isBull, string theme, color bullCol, color bearCol) =>
    monoBull = #9CA3AF
    monoBear = #4B5563
    theme == "Monochrome" ? (isBull ? monoBull : monoBear) : (isBull ? bullCol : bearCol)

f_isInPrioritySession() =>
    bool result = true
    if prioritySessionInput != "None"
        h = hour(time, "UTC")
        result := switch prioritySessionInput
            "London" => h >= 7 and h < 10
            "New York" => h >= 12 and h < 15
            "Asia" => h >= 0 and h < 8
            => true
    result

// -- Internal engine: Structure --------------------------------------------------
f_updateStructureSnapshot(StructureSnapshot st, float atrVal) =>
    ph = ta.pivothigh(swingLenInput, swingLenInput)
    pl = ta.pivotlow(swingLenInput, swingLenInput)
    if not na(ph)
        st.pivotHigh := ph
    if not na(pl)
        st.pivotLow := pl
    
    st.bullishBOS := false
    st.bearishBOS := false
    st.bullishCHoCH := false
    st.bearishCHoCH := false
    
    checkSrcUp = confirmModeInput == "Wick" ? high : close
    checkSrcDown = confirmModeInput == "Wick" ? low : close
    
    breakUp = not na(st.pivotHigh) and checkSrcUp > st.pivotHigh + atrVal * 0.25
    breakDown = not na(st.pivotLow) and checkSrcDown < st.pivotLow - atrVal * 0.25
    
    if breakUp
        if st.bias < 0
            st.bullishCHoCH := true
        else
            st.bullishBOS := true
        st.bias := 1
        
    if breakDown
        if st.bias > 0
            st.bearishCHoCH := true
        else
            st.bearishBOS := true
        st.bias := -1
        
    if consolidationCheckInput
        rangeHigh = ta.highest(high, consolidationLenInput)
        rangeLow = ta.lowest(low, consolidationLenInput)
        st.isConsolidating := (rangeHigh - rangeLow) <= consolidationAtrMultInput * atrVal
    else
        st.isConsolidating := false

// -- Internal engine: Liquidity ---------------------------------------------------
f_updateLiquiditySnapshot(LiquiditySnapshot liq, StructureSnapshot st, float atrVal) =>
    liq.recentSweep := false
    if not na(st.pivotLow) and low < st.pivotLow and close > st.pivotLow
        liq.recentSweep := true
        liq.sweepDirection := 1
        liq.sweepType := "Bullish"
    if not na(st.pivotHigh) and high > st.pivotHigh and close < st.pivotHigh
        liq.recentSweep := true
        liq.sweepDirection := -1
        liq.sweepType := "Bearish"
    liq.eqhPresent := math.abs(high - ta.highest(high, 50)) < atrVal * 0.1
    liq.eqlPresent := math.abs(low - ta.lowest(low, 50)) < atrVal * 0.1

// -- Internal engine: Order Block -------------------------------------------------
f_calculateOBQuality(float rangeSize, float atrVal) =>
    ratio = atrVal > 0 ? rangeSize / atrVal : 0.0
    string q = ratio >= 3.0 ? "Very Strong" : ratio >= 2.0 ? "Strong" : ratio >= 1.0 ? "Normal" : "Weak"
    q

f_getOBQualityScore(string quality) =>
    float s = switch quality
        "Very Strong" => 90.0
        "Strong" => 70.0
        "Normal" => 45.0
        => 20.0
    s

f_determineOBState(float top, float bottom, bool isBullish) =>
    mid = (top + bottom) / 2.0
    string state = "Active"
    if isBullish
        if close >= top
            state := "Reclaimed"
        else if close <= bottom
            state := "Breaker"
        else if close > mid
            state := "Mitigated"
    else
        if close <= bottom
            state := "Reclaimed"
        else if close >= top
            state := "Breaker"
        else if close < mid
            state := "Mitigated"
    state

f_updateOrderBlockSnapshot(OrderBlockSnapshot ob, StructureSnapshot st, float atrVal) =>
    if st.bullishBOS or st.bullishCHoCH
        for i = 1 to 20
            if close[i] < open[i]
                ob.obTop := high[i]
                ob.obBottom := low[i]
                ob.obQualityText := f_calculateOBQuality(high[i] - low[i], atrVal)
                ob.exists := true
                break
    if st.bearishBOS or st.bearishCHoCH
        for i = 1 to 20
            if close[i] > open[i]
                ob.obTop := high[i]
                ob.obBottom := low[i]
                ob.obQualityText := f_calculateOBQuality(high[i] - low[i], atrVal)
                ob.exists := true
                break
    if ob.exists
        ob.obState := f_determineOBState(ob.obTop, ob.obBottom, st.bias == 1)

// -- Internal engine: FVG ---------------------------------------------------------
f_updateFVGSnapshot(FVGSnapshot fvg) =>
    if low > high[2]
        fvg.fvgTop := low
        fvg.fvgBottom := high[2]
        fvg.fvgExists := true
        fvg.bullish := true
        fvg.fvgState := "Active"
    else if high < low[2]
        fvg.fvgTop := low[2]
        fvg.fvgBottom := high
        fvg.fvgExists := true
        fvg.bullish := false
        fvg.fvgState := "Active"
    if fvg.fvgExists
        if fvg.bullish and close <= fvg.fvgBottom
            fvg.fvgState := "Mitigated"
        if not fvg.bullish and close >= fvg.fvgTop
            fvg.fvgState := "Mitigated"

// -- Zones -----------------------------------------------------------------------
f_updateZoneSnapshot(ZoneSnapshot z, StructureSnapshot st) =>
    if not premiumDiscountCalcInput
        z.currentZone := "N/A"
    else if na(st.pivotHigh) or na(st.pivotLow)
        z.currentZone := "N/A"
    else
        rangeTop = st.pivotHigh
        rangeBot = st.pivotLow
        eqLevel = (rangeTop + rangeBot) / 2.0
        oteTop = rangeTop - (rangeTop - rangeBot) * oteStartInput
        oteBot = rangeTop - (rangeTop - rangeBot) * oteEndInput
        goldenTop = rangeTop - (rangeTop - rangeBot) * goldenStartInput
        goldenBot = rangeTop - (rangeTop - rangeBot) * goldenEndInput
        if close > rangeTop
            z.currentZone := "Above"
        else if close < rangeBot
            z.currentZone := "Below"
        else if close >= goldenTop and close <= goldenBot
            z.currentZone := "Golden"
        else if close >= oteTop and close <= oteBot
            z.currentZone := "OTE"
        else if close > eqLevel
            z.currentZone := "Premium"
        else
            z.currentZone := "Discount"

// -- Scoring Modules --------------------------------------------------------------
f_module1_structureQuality(StructureSnapshot st) =>
    score = 0.0
    detail = ""
    if st.bias != 0
        score += 40
        detail := detail + "Clear bias "
    if not st.isConsolidating
        score += 25
        detail := detail + "| Not consolidating "
    else
        detail := detail + "| Consolidating "
    if st.bullishBOS or st.bullishCHoCH or st.bearishBOS or st.bearishCHoCH
        score += 20
        detail := detail + "| Recent break"
    ModuleScore.new(score = math.min(score, 100), detail = detail, passed = score >= 50)

f_module2_bosQuality(StructureSnapshot st, LiquiditySnapshot liq) =>
    score = 0.0
    detail = ""
    if st.bullishBOS
        score := 70
        detail := "Bullish BOS | Continuation"
    else if st.bearishBOS
        score := 70
        detail := "Bearish BOS | Continuation"
    else if st.bullishCHoCH
        score := 55
        detail := "Bullish CHoCH | Reversal"
        if liq.recentSweep and liq.sweepDirection == -1
            score += 15
            detail := detail + " + Sweep"
    else if st.bearishCHoCH
        score := 55
        detail := "Bearish CHoCH | Reversal"
        if liq.recentSweep and liq.sweepDirection == 1
            score += 15
            detail := detail + " + Sweep"
    else
        detail := "No recent break"
    ModuleScore.new(score = math.min(score, 100), detail = detail, passed = score >= 40)

f_module3_liquidity(LiquiditySnapshot liq) =>
    score = 0.0
    detail = ""
    if liq.recentSweep
        score += 50
        detail := "Confirmed sweep " + liq.sweepType
    if liq.eqhPresent
        score += 15
        detail := detail + "| EQH nearby"
    if liq.eqlPresent
        score += 15
        detail := detail + "| EQL nearby"
    if score == 0
        detail := "No liquidity detected"
    ModuleScore.new(score = math.min(score, 100), detail = detail, passed = score >= 30)

f_module4_orderBlockQuality(OrderBlockSnapshot ob) =>
    score = 0.0
    detail = ""
    if ob.exists
        score := f_getOBQualityScore(ob.obQualityText)
        detail := ob.obQualityText + " | " + ob.obState
        if ob.obState == "Active"
            score += 10
    else
        detail := "No nearby Order Block"
    ModuleScore.new(score = math.min(score, 100), detail = detail, passed = score >= 40)

f_module5_fvgQuality(FVGSnapshot fvg, float atrVal) =>
    score = 0.0
    detail = ""
    if fvg.fvgExists
        gapSize = fvg.fvgTop - fvg.fvgBottom
        gapRatio = atrVal > 0 ? gapSize / atrVal : 0.0
        score := math.min(gapRatio * 15, 80)
        detail := fvg.fvgState + " | Gap: " + str.tostring(gapRatio, "#.00") + "x ATR"
        if fvg.fvgState == "Active"
            score += 10
    else
        detail := "No nearby FVG"
    ModuleScore.new(score = math.min(score, 100), detail = detail, passed = score >= 30)

f_module6_zone(ZoneSnapshot z) =>
    score = switch z.currentZone
        "Golden" => 100.0
        "OTE" => 85.0
        "Premium" => 55.0
        "Discount" => 55.0
        "Above" => 20.0
        "Below" => 20.0
        => 30.0
    detail = switch z.currentZone
        "Golden" => "Golden Zone"
        "OTE" => "OTE Zone"
        "Premium" => "Premium"
        "Discount" => "Discount"
        "Above" => "Above range"
        "Below" => "Below range"
        => "Undetermined zone"
    ModuleScore.new(score = score, detail = detail, passed = score >= 40)

f_module7_mtfAlignment(StructureSnapshot st, int htfBias) =>
    score = 0.0
    detail = ""
    if htfBias == st.bias and htfBias != 0
        score := 100
        detail := "HTF align: " + (htfBias == 1 ? "Bull" : "Bear")
    else if htfBias == 0
        score := 60
        detail := "HTF neutral"
    else
        score := 0
        detail := "HTF conflicts"
    ModuleScore.new(score = score, detail = detail, passed = score >= 50)

f_module8_session() =>
    score = 0.0
    detail = ""
    if not showSessionsInput
        score := 50
        detail := "Sessions disabled"
    else if f_isInPrioritySession()
        score := 100
        detail := "In priority session"
    else
        score := 60
        detail := "Outside priority session"
    ModuleScore.new(score = score, detail = detail, passed = true)

f_module9_riskReward(OrderBlockSnapshot ob, FVGSnapshot fvg) =>
    rr = 1.5
    if ob.exists and fvg.fvgExists
        distToInvalidation = math.abs(close - ob.obBottom)
        distToTarget = math.abs(fvg.fvgTop - close)
        if distToInvalidation > 0
            rr := distToTarget / distToInvalidation
    score = 0.0
    detail = ""
    if rr >= 3.0
        score := 100
        detail := "RR " + str.tostring(rr, "#.00") + " | Excellent"
    else if rr >= 2.0
        score := 80
        detail := "RR " + str.tostring(rr, "#.00") + " | Good"
    else if rr >= 1.5
        score := 60
        detail := "RR " + str.tostring(rr, "#.00") + " | Acceptable"
    else
        score := 30
        detail := "RR " + str.tostring(rr, "#.00") + " | Poor"
    ModuleScore.new(score = score, detail = detail, passed = rr >= rrMinimumInput)

f_module10_regimeMultiplier(StructureSnapshot st) =>
    float mult = 1.0
    if st.isConsolidating
        mult := 0.75
    else if st.bullishBOS or st.bearishBOS
        mult := 1.05
    else if st.bullishCHoCH or st.bearishCHoCH
        mult := 0.85
    mult

f_normalizeWeights(float w1, float w2, float w3, float w4, float w5, float w6, float w7, float w8, float w9) =>
    total = w1 + w2 + w3 + w4 + w5 + w6 + w7 + w8 + w9
    if total > 0
        [w1 / total * 100.0, w2 / total * 100.0, w3 / total * 100.0, w4 / total * 100.0, w5 / total * 100.0, w6 / total * 100.0, w7 / total * 100.0, w8 / total * 100.0, w9 / total * 100.0]
    else
        [w1, w2, w3, w4, w5, w6, w7, w8, w9]

f_checkGates(StructureSnapshot st, LiquiditySnapshot liq, OrderBlockSnapshot ob, FVGSnapshot fvg, int htfBias, bool rrPassed) =>
    maxCap = 100.0
    failedGates = 0
    gateDetails = ""
    if gateConflictInput and htfBias != 0 and htfBias != st.bias
        failedGates += 1
        maxCap := math.min(maxCap, 50.0)
        gateDetails := gateDetails + "❌ HTF conflicts\n"
    if gateAbsentLevelInput and not ob.exists and not fvg.fvgExists
        failedGates += 1
        maxCap := math.min(maxCap, 65.0)
        gateDetails := gateDetails + "❌ No nearby level\n"
    if gateSweptLiquidityInput and (st.bullishCHoCH or st.bearishCHoCH) and not liq.recentSweep
        failedGates += 1
        maxCap := math.min(maxCap, 70.0)
        gateDetails := gateDetails + "❌ CHoCH without sweep\n"
    if gateRRInput and not rrPassed
        failedGates += 1
        maxCap := math.min(maxCap, 60.0)
        gateDetails := gateDetails + "❌ Insufficient RR\n"
    [failedGates, maxCap, gateDetails]

// -- Dashboard -----------------------------------------------------------------
f_populateDashboard(table t, ScoringResult result, StructureSnapshot st, ZoneSnapshot z) =>
    noteColor = (result.notation == "A+" or result.notation == "A") ? bullColorInput : (result.notation == "B" or result.notation == "C") ? color.orange : bearColorInput
    filledBlocks = int(result.scoreFinal / 10.0)
    progressBar = ""
    for i = 0 to 9
        progressBar := progressBar + (i < filledBlocks ? "█" : "░")
    
    row = 0
    table.cell(t, 0, row, "VERISCOPE INTELLIGENCE", text_color = color.white, text_size = size.small, bgcolor = color.new(color.black, 0), text_halign = text.align_center)
    table.cell(t, 1, row, "", bgcolor = color.new(color.black, 0))
    
    row += 1
    table.cell(t, 0, row, "Score", text_color = color.gray, text_size = size.small)
    table.cell(t, 1, row, str.tostring(result.scoreFinal, "#.0") + "/100 | " + result.notation, text_color = noteColor, text_size = size.small, text_halign = text.align_right)
    
    row += 1
    table.cell(t, 0, row, "Progress", text_color = color.gray, text_size = size.small)
    table.cell(t, 1, row, progressBar, text_color = color.white, text_size = size.small, text_halign = text.align_right)
    
    row += 1
    directionText = st.bias == 1 ? "🔼 BULLISH" : st.bias == -1 ? "🔽 BEARISH" : "◼ NEUTRAL"
    directionColor = st.bias == 1 ? bullColorInput : st.bias == -1 ? bearColorInput : color.gray
    table.cell(t, 0, row, "Direction", text_color = color.gray, text_size = size.small)
    table.cell(t, 1, row, directionText, text_color = directionColor, text_size = size.small, text_halign = text.align_right)
    
    row += 1
    regimeText = st.isConsolidating ? "CONSOLIDATION" : "TREND"
    table.cell(t, 0, row, "Regime", text_color = color.gray, text_size = size.small)
    table.cell(t, 1, row, regimeText, text_color = st.isConsolidating ? color.orange : color.white, text_size = size.small, text_halign = text.align_right)
    
    row += 1
    table.cell(t, 0, row, "Zone", text_color = color.gray, text_size = size.small)
    table.cell(t, 1, row, z.currentZone, text_color = color.white, text_size = size.small, text_halign = text.align_right)
    
    row += 1

f_themeColor(bool isBull, string theme, color bullCol, color bearCol) =>
    monoBull = #9CA3AF
    monoBear = #4B5563
    theme == "Monochrome" ? (isBull ? monoBull : monoBear) : (isBull ? bullCol : bearCol)

f_dashPosition(string p) =>
    switch p
        "Top Right"    => position.top_right
        "Top Left"     => position.top_left
        "Bottom Right" => position.bottom_right
        "Bottom Left"  => position.bottom_left
        => position.top_right

f_biasText(int b) =>
    b == 1 ? "Bullish" : b == -1 ? "Bearish" : "Neutral"

f_biasColor(int b, color bull, color bear) =>
    b == 1 ? bull : b == -1 ? bear : color.gray

f_sweepLabel(bool stopHunt, string strength) =>
    stopHunt ? "Stop Hunt" : (strength == "Strong" ? "Liq. Grab" : "Sweep")

f_isBreakUp(float level, float atrVal) =>
    candleRange = high - low
    penetration = candleRange > 0 ? (close - level) / candleRange * 100.0 : 0.0
    baseResult = switch confirmModeInput
        "Close"              => close > level
        "Wick"                => high > level
        "Candle Percentage"   => close > level and penetration >= candleBodyPctInput
        "ATR"                 => close > level + atrDistMultInput * atrVal
        => close > level
    checkSrc = confirmModeInput == "Wick" ? high : close
    baseResult and checkSrc > level + extraMinDistInput * atrVal

f_isBreakDown(float level, float atrVal) =>
    candleRange = high - low
    penetration = candleRange > 0 ? (level - close) / candleRange * 100.0 : 0.0
    baseResult = switch confirmModeInput
        "Close"              => close < level
        "Wick"                => low < level
        "Candle Percentage"   => close < level and penetration >= candleBodyPctInput
        "ATR"                 => close < level - atrDistMultInput * atrVal
        => close < level
    checkSrc = confirmModeInput == "Wick" ? low : close
    baseResult and checkSrc < level - extraMinDistInput * atrVal

f_updateStructure(StructureState st, int len, float atrVal, float eqTol, float strengthMult, bool showPoints, color bullCol, color bearCol) =>
    st.bullishBOS   := false
    st.bullishCHoCH := false
    st.bearishBOS   := false
    st.bearishCHoCH := false
    st.newEQH       := false
    st.newEQL       := false
    ph = ta.pivothigh(len, len)
    pl = ta.pivotlow(len, len)
    if not na(ph)
        pBar = bar_index - len
        impulseHigh = math.abs(close - ph)
        newStrengthHigh = impulseHigh >= strengthMult * atrVal ? "Strong" : "Weak"
        if not na(st.pivotHighPrice)
            if math.abs(ph - st.pivotHighPrice) <= eqTol * atrVal
                st.newEQH       := true
                st.eqHighPrice1 := st.pivotHighPrice
                st.eqHighBar1   := st.pivotHighBar
            if showPoints
                tagHigh = ph > st.pivotHighPrice ? "HH" : "LH"
                label.new(pBar, ph, tagHigh, style = label.style_label_down, textcolor = bearCol, color = color.new(color.black, 100), size = size.tiny)
        if not st.lowBroken
            st.lowProtected := true
        st.pivotHighPrice := ph
        st.pivotHighBar   := pBar
        st.highBroken     := false
        st.highStrength   := newStrengthHigh
        st.highProtected  := false
    if not na(pl)
        pBar2 = bar_index - len
        impulseLow = math.abs(close - pl)
        newStrengthLow = impulseLow >= strengthMult * atrVal ? "Strong" : "Weak"
        if not na(st.pivotLowPrice)
            if math.abs(pl - st.pivotLowPrice) <= eqTol * atrVal
                st.newEQL      := true
                st.eqLowPrice1 := st.pivotLowPrice
                st.eqLowBar1   := st.pivotLowBar
            if showPoints
                tagLow = pl < st.pivotLowPrice ? "LL" : "HL"
                label.new(pBar2, pl, tagLow, style = label.style_label_up, textcolor = bullCol, color = color.new(color.black, 100), size = size.tiny)
        if not st.highBroken
            st.highProtected := true
        st.pivotLowPrice := pl
        st.pivotLowBar   := pBar2
        st.lowBroken     := false
        st.lowStrength   := newStrengthLow
        st.lowProtected  := false
    if not st.highBroken and not na(st.pivotHighPrice) and f_isBreakUp(st.pivotHighPrice, atrVal)
        isChochUp = st.bias < 0
        st.highBroken := true
        st.bias := 1
        if isChochUp
            st.bullishCHoCH := true
        else
            st.bullishBOS := true
    if not st.lowBroken and not na(st.pivotLowPrice) and f_isBreakDown(st.pivotLowPrice, atrVal)
        isChochDn = st.bias > 0
        st.lowBroken := true
        st.bias := -1
        if isChochDn
            st.bearishCHoCH := true
        else
            st.bearishBOS := true

f_prevLevelLine(bool show, float level, color col, string tag) =>
    var line lineId    = na
    var label lbl      = na
    var float lastLevel = na
    if show and not na(level)
        if na(lineId) or level != lastLevel
            if not na(lineId)
                line.delete(lineId)
            if not na(lbl)
                label.delete(lbl)
            lineId := line.new(bar_index, level, bar_index, level, color = col, style = line.style_dashed, extend = extend.none)
            lbl    := label.new(bar_index, level, tag, style = label.style_label_left, textcolor = col, color = color.new(color.black, 100), size = size.tiny)
            lastLevel := level
        else
            line.set_xy2(lineId, bar_index, level)
            label.set_x(lbl, bar_index)
            label.set_y(lbl, level)
    else if not show and not na(lineId)
        line.delete(lineId)
        label.delete(lbl)
        lineId := na
        lbl := na
    lineId

f_sessionRange(bool inSession, bool show, color col) =>
    var float sessHigh = na
    var float sessLow  = na
    var box   sessBox  = na
    var bool  wasIn    = false
    if inSession
        if not wasIn
            sessHigh := high
            sessLow  := low
            if show
                sessBox := box.new(bar_index, high, bar_index, low, border_color = color.new(col, 50), bgcolor = color.new(col, 92), extend = extend.none)
        else
            sessHigh := math.max(sessHigh, high)
            sessLow  := math.min(sessLow, low)
            if show and not na(sessBox)
                box.set_top(sessBox, sessHigh)
                box.set_bottom(sessBox, sessLow)
                box.set_right(sessBox, bar_index)
    else
        if wasIn and not na(sessBox)
            box.set_right(sessBox, bar_index)
            sessBox := na
    wasIn := inSession
    [sessHigh, sessLow]

f_createOB(bool bullish, int lookback) =>
    float obTop  = na
    float obBot  = na
    int   obBar  = na
    float displacement = na
    if bullish
        for i = 1 to lookback
            if close[i] < open[i]
                obTop := high[i]
                obBot := low[i]
                obBar := bar_index - i
                displacement := close - close[i]
                break
    else
        for i = 1 to lookback
            if close[i] > open[i]
                obTop := high[i]
                obBot := low[i]
                obBar := bar_index - i
                displacement := close[i] - close
                break
    [obTop, obBot, obBar, displacement]

f_obQuality(float displacement, float atrVal) =>
    ratio = atrVal > 0 ? displacement / atrVal : 0.0
    string q = switch
        ratio >= qualityStrongMultInput => "Very Strong"
        ratio >= qualityGoodMultInput   => "Strong"
        ratio >= qualityNormalMultInput => "Normal"
        => "Weak"
    q

f_obLabelText(OrderBlock ob) =>
    base = ob.bullish ? "OB+" : "OB-"
    switch ob.state
        "Breaker"   => "Breaker"
        "Reclaimed" => "Reclaimed"
        => base + " (" + ob.quality + ")"

f_priceZoneStatus(float p, float rangeTop, float rangeBot, float oteTop, float oteBot, float gTop, float gBot) =>
    if p > rangeTop
        "Above"
    else if p < rangeBot
        "Below"
    else
        eqLevel = (rangeTop + rangeBot) / 2.0
        if p > eqLevel
            if p >= oteTop and p <= oteBot
                "OTE"
            else if p >= gTop and p <= gBot
                "Golden"
            else
                "Premium"
        else
            if p >= oteTop and p <= oteBot
                "OTE"
            else if p >= gTop and p <= gBot
                "Golden"
            else
                "Discount"

//@version=6
// =============================================================================
// VERISCOPE ALERT ENGINE v1.2 — Autonomous Confluence Watchdog
// Fix 1: guarded all "for i = 0 to N-1" loops against N=0 (RE10045 on bar 0
//        when arrays are empty — Pine loops from 0 to -1 still execute once).
// Fix 2: Alert History table now sizes its rows dynamically from maxHistory
//        instead of a fixed 12, preventing out-of-bounds errors when the
//        user sets "Max Alerts Shown in History" above 11.
// =============================================================================

indicator("Veriscope Alert Engine", shorttitle = "Alert Engine", overlay = true,
     max_boxes_count = 200, max_lines_count = 200, max_labels_count = 200)

GRP_LANG = "Language / Idioma"
langInput = input.string("English", "Dashboard Language", options = ["English", "Português"], group = GRP_LANG)

f_t(string en, string pt) =>
    langInput == "Português" ? pt : en

f_pos(string p) =>
    switch p
        "Top Right"    => position.top_right
        "Top Left"     => position.top_left
        "Bottom Right" => position.bottom_right
        "Bottom Left"  => position.bottom_left
        => position.top_right

GRP_RULE = "Alert Rule Builder — Criteria"
useLiquiditySweep = input.bool(true,  "Use: Liquidity Sweep", group = GRP_RULE)
useStructure      = input.bool(true,  "Use: BOS / CHoCH Confirmed", group = GRP_RULE)
useFVG            = input.bool(true,  "Use: Price Inside Valid FVG", group = GRP_RULE)
usePremDisc       = input.bool(true,  "Use: Premium / Discount Zone", group = GRP_RULE)
useSessionFilter  = input.bool(true,  "Use: Session Filter", group = GRP_RULE)
useVolatility     = input.bool(false, "Use: Volatility Expansion", group = GRP_RULE)
useScore          = input.bool(false, "Use: External Score (Veriscope Intelligence)", group = GRP_RULE)

GRP_LOGIC = "Alert Rule Builder — Match Logic"
matchMode      = input.string("All (AND)", "Match Requirement", options = ["All (AND)", "At least N of M"], group = GRP_LOGIC)
minMatchesIn   = input.int(3, "N (if 'At least N of M')", minval = 1, group = GRP_LOGIC)
cooldownBars   = input.int(3, "Cooldown Between Alerts (bars)", minval = 1, group = GRP_LOGIC)
mediumRatioIn  = input.float(0.6, "Medium Priority Threshold (ratio of matched criteria)", minval = 0.1, maxval = 0.99, step = 0.05, group = GRP_LOGIC)

GRP_LIQ = "Liquidity Sweep Settings"
swingLen = input.int(5, "Swing Lookback (bars each side)", minval = 2, group = GRP_LIQ)

swH = ta.pivothigh(high, swingLen, swingLen)
swL = ta.pivotlow(low, swingLen, swingLen)

var float lastSwingHigh = na
var float lastSwingLow  = na
prevSwingHigh = lastSwingHigh
prevSwingLow  = lastSwingLow
if not na(swH)
    lastSwingHigh := swH
if not na(swL)
    lastSwingLow := swL

sweepHigh = not na(prevSwingHigh) and high > prevSwingHigh and close < prevSwingHigh
sweepLow  = not na(prevSwingLow) and low < prevSwingLow and close > prevSwingLow
liquiditySwept = sweepHigh or sweepLow

GRP_STR = "Structure (BOS/CHoCH) Settings"
structureValidBars = input.int(10, "Structure Stays 'Confirmed' For (bars)", minval = 1, group = GRP_STR)

var int bias = 0
var int lastStructureBar = na
breakUp   = not na(lastSwingHigh) and ta.crossover(close, lastSwingHigh)
breakDown = not na(lastSwingLow) and ta.crossunder(close, lastSwingLow)

if breakUp
    lastStructureBar := bar_index
    bias := 1
if breakDown
    lastStructureBar := bar_index
    bias := -1

structureRecentlyConfirmed = not na(lastStructureBar) and (bar_index - lastStructureBar) <= structureValidBars

GRP_FVG = "Fair Value Gap Settings"
maxFVG        = input.int(20, "Max Tracked FVGs", minval = 5, group = GRP_FVG)
showFVGBoxes  = input.bool(true, "Draw FVG Boxes", group = GRP_FVG)

bullishFVG = low > high[2]
bearishFVG = high < low[2]

var array<float> fvgTop = array.new<float>()
var array<float> fvgBot = array.new<float>()
var array<bool>  fvgBull = array.new<bool>()
var array<bool>  fvgMit  = array.new<bool>()
var array<box>   fvgBox  = array.new<box>()

if bullishFVG
    array.unshift(fvgTop, low)
    array.unshift(fvgBot, high[2])
    array.unshift(fvgBull, true)
    array.unshift(fvgMit, false)
    newBox = showFVGBoxes ? box.new(bar_index[2], low, bar_index, high[2], border_color = color.new(color.teal, 40), bgcolor = color.new(color.teal, 88), extend = extend.right) : box(na)
    array.unshift(fvgBox, newBox)
    if array.size(fvgTop) > maxFVG
        array.pop(fvgTop)
        array.pop(fvgBot)
        array.pop(fvgBull)
        array.pop(fvgMit)
        oldBox = array.pop(fvgBox)
        if not na(oldBox)
            box.delete(oldBox)

if bearishFVG
    array.unshift(fvgTop, low[2])
    array.unshift(fvgBot, high)
    array.unshift(fvgBull, false)
    array.unshift(fvgMit, false)
    newBox2 = showFVGBoxes ? box.new(bar_index[2], low[2], bar_index, high, border_color = color.new(color.red, 40), bgcolor = color.new(color.red, 88), extend = extend.right) : box(na)
    array.unshift(fvgBox, newBox2)
    if array.size(fvgTop) > maxFVG
        array.pop(fvgTop)
        array.pop(fvgBot)
        array.pop(fvgBull)
        array.pop(fvgMit)
        oldBox2 = array.pop(fvgBox)
        if not na(oldBox2)
            box.delete(oldBox2)

priceInsideValidFVG = false
if array.size(fvgTop) > 0
    for i = 0 to array.size(fvgTop) - 1
        if not array.get(fvgMit, i)
            top = array.get(fvgTop, i)
            bot = array.get(fvgBot, i)
            isBull = array.get(fvgBull, i)
            bx = array.get(fvgBox, i)
            if isBull
                if close < bot
                    array.set(fvgMit, i, true)
                    if not na(bx)
                        box.set_extend(bx, extend.none)
                else if close <= top and close >= bot
                    priceInsideValidFVG := true
            else
                if close > top
                    array.set(fvgMit, i, true)
                    if not na(bx)
                        box.set_extend(bx, extend.none)
                else if close <= top and close >= bot
                    priceInsideValidFVG := true

GRP_PD = "Premium / Discount Settings"
pdMode = input.string("Either side", "Required Zone", options = ["Discount only", "Premium only", "Either side"], group = GRP_PD)

rangeMid = (not na(lastSwingHigh) and not na(lastSwingLow)) ? (lastSwingHigh + lastSwingLow) / 2.0 : na
inDiscount = not na(rangeMid) and close < rangeMid
inPremium  = not na(rangeMid) and close > rangeMid
pdOK = pdMode == "Discount only" ? inDiscount : pdMode == "Premium only" ? inPremium : (inDiscount or inPremium)

GRP_SESS = "Session Filter (DST-aware)"
sessionChoice = input.string("London", "Required Session", options = ["Asia", "London", "New York", "London x NY Overlap", "Custom"], group = GRP_SESS)
asiaSession   = input.session("0000-0900", "Asia Session", group = GRP_SESS)
asiaTZ        = input.string("Asia/Tokyo", "Asia Timezone", group = GRP_SESS)
londonSession = input.session("0700-1600", "London Session", group = GRP_SESS)
londonTZ      = input.string("Europe/London", "London Timezone", group = GRP_SESS)
nySession     = input.session("0800-1700", "New York Session", group = GRP_SESS)
nyTZ          = input.string("America/New_York", "New York Timezone", group = GRP_SESS)
customSession = input.session("0000-2359", "Custom Session", group = GRP_SESS)
customTZ      = input.string("UTC", "Custom Timezone", group = GRP_SESS)

inAsiaS   = not na(time(timeframe.period, asiaSession, asiaTZ))
inLondonS = not na(time(timeframe.period, londonSession, londonTZ))
inNYS     = not na(time(timeframe.period, nySession, nyTZ))
inCustomS = not na(time(timeframe.period, customSession, customTZ))
inOverlapS = inLondonS and inNYS

sessionOK = sessionChoice == "Asia" ? inAsiaS : sessionChoice == "London" ? inLondonS : sessionChoice == "New York" ? inNYS : sessionChoice == "London x NY Overlap" ? inOverlapS : inCustomS

GRP_VOL = "Volatility Filter"
volMult = input.float(1.2, "Alert only if ATR >= x times its average", minval = 1.0, step = 0.1, group = GRP_VOL)

atrNow = ta.atr(14)
atrAvg = ta.sma(atrNow, 50)
volatilityOK = atrNow >= atrAvg * volMult

GRP_SCORE = "External Score (Veriscope Intelligence)"
scoreSource    = input.source(close, "Score Source (map to Intelligence plot)", group = GRP_SCORE)
scoreThreshold = input.int(70, "Minimum Score", minval = 0, maxval = 100, group = GRP_SCORE)
scoreOK = scoreSource >= scoreThreshold

names  = array.new<string>()
states = array.new<bool>()

if useLiquiditySweep
    array.push(names, f_t("Liquidity Sweep", "Varrimento de Liquidez"))
    array.push(states, liquiditySwept)
if useStructure
    array.push(names, "BOS / CHoCH")
    array.push(states, structureRecentlyConfirmed)
if useFVG
    array.push(names, f_t("Price in Valid FVG", "Preço em FVG Válido"))
    array.push(states, priceInsideValidFVG)
if usePremDisc
    array.push(names, f_t("Premium/Discount Zone", "Zona Premium/Discount"))
    array.push(states, pdOK)
if useSessionFilter
    array.push(names, f_t("Session Filter", "Filtro de Sessão"))
    array.push(states, sessionOK)
if useVolatility
    array.push(names, f_t("Volatility Expansion", "Expansão de Volatilidade"))
    array.push(states, volatilityOK)
if useScore
    array.push(names, f_t("Intelligence Score", "Score Intelligence"))
    array.push(states, scoreOK)

enabledCount = array.size(states)
trueCount = 0
if enabledCount > 0
    for i = 0 to enabledCount - 1
        if array.get(states, i)
            trueCount += 1

requiredMet = enabledCount == 0 ? false : (matchMode == "All (AND)" ? trueCount == enabledCount : trueCount >= math.min(minMatchesIn, enabledCount))

ratio = enabledCount > 0 ? trueCount / enabledCount : 0.0
priorityLabel = enabledCount == 0 ? "—" : (ratio >= 1.0 ? "HIGH" : ratio >= mediumRatioIn ? "MEDIUM" : ratio > 0 ? "LOW" : "—")

var int lastAlertBar = na
cooldownRemaining = na(lastAlertBar) ? 0 : math.max(0, cooldownBars - (bar_index - lastAlertBar))
triggerNow = requiredMet and (na(lastAlertBar) or (bar_index - lastAlertBar) >= cooldownBars)

GRP_HIST = "Alert History Panel"
maxHistory = input.int(8, "Max Alerts Shown in History", minval = 3, maxval = 20, group = GRP_HIST)
var array<string> historyLog = array.new<string>()

matchedListStr = ""
if enabledCount > 0
    for i = 0 to enabledCount - 1
        if array.get(states, i)
            matchedListStr := matchedListStr + (matchedListStr == "" ? "" : " | ") + array.get(names, i)

if triggerNow
    array.unshift(historyLog, str.format_time(time, "dd MMM HH:mm") + " | " + priorityLabel + " | " + str.tostring(close, format.mintick))
    if array.size(historyLog) > maxHistory
        array.pop(historyLog)
    msg = f_t("VERISCOPE ALERT ENGINE\n", "VERISCOPE ALERT ENGINE\n") + syminfo.ticker + " (" + timeframe.period + ")\n" + f_t("Priority: ", "Prioridade: ") + priorityLabel + "\n" + f_t("Matched: ", "Critérios: ") + matchedListStr + "\n" + f_t("Open the chart and analyze. Not a trade signal.", "Abra o gráfico e analise. Isto não é um sinal de compra/venda.")
    alert(msg, alert.freq_once_per_bar_close)
    lastAlertBar := bar_index

alertcondition(triggerNow, "Veriscope Alert Engine — Any Match", "Veriscope Alert Engine: rule conditions met. Open the chart and analyze. Not a trade signal.")
alertcondition(triggerNow and priorityLabel == "HIGH", "Veriscope Alert Engine — HIGH Priority", "Veriscope Alert Engine: HIGH priority confluence. Open the chart and analyze. Not a trade signal.")
alertcondition(triggerNow and priorityLabel == "MEDIUM", "Veriscope Alert Engine — MEDIUM Priority", "Veriscope Alert Engine: MEDIUM priority confluence. Open the chart and analyze. Not a trade signal.")
alertcondition(triggerNow and priorityLabel == "LOW", "Veriscope Alert Engine — LOW Priority", "Veriscope Alert Engine: LOW priority confluence. Open the chart and analyze. Not a trade signal.")

GRP_VIS = "Dashboard & Visuals"
showDashboard = input.bool(true, "Show Dashboard", group = GRP_VIS)
dashPos       = input.string("Top Right", "Dashboard Position", options = ["Top Right", "Top Left", "Bottom Right", "Bottom Left"], group = GRP_VIS)
showHistPanel = input.bool(true, "Show History Panel", group = GRP_VIS)
histPos       = input.string("Bottom Right", "History Panel Position", options = ["Top Right", "Top Left", "Bottom Right", "Bottom Left"], group = GRP_VIS)
showPDLines   = input.bool(false, "Show Premium/Discount Range", group = GRP_VIS)

var line pdTopLine = na
var line pdMidLine = na
var line pdBotLine = na
if showPDLines and not na(lastSwingHigh) and not na(lastSwingLow)
    if na(pdTopLine)
        pdTopLine := line.new(bar_index, lastSwingHigh, bar_index, lastSwingHigh, color = color.new(color.gray, 40))
        pdMidLine := line.new(bar_index, rangeMid, bar_index, rangeMid, color = color.new(color.gray, 60), style = line.style_dashed)
        pdBotLine := line.new(bar_index, lastSwingLow, bar_index, lastSwingLow, color = color.new(color.gray, 40))
    else
        line.set_xy1(pdTopLine, bar_index - 1, lastSwingHigh)
        line.set_xy2(pdTopLine, bar_index, lastSwingHigh)
        line.set_xy1(pdMidLine, bar_index - 1, rangeMid)
        line.set_xy2(pdMidLine, bar_index, rangeMid)
        line.set_xy1(pdBotLine, bar_index - 1, lastSwingLow)
        line.set_xy2(pdBotLine, bar_index, lastSwingLow)

if showDashboard and barstate.islast
    var table dash = table.new(f_pos(dashPos), 2, 20, border_width = 1, bgcolor = color.new(color.black, 15))
    r = 0
    table.cell(dash, 0, r, "VERISCOPE ALERT ENGINE", text_color = color.white, bgcolor = color.new(color.black, 0), text_size = size.small)
    table.cell(dash, 1, r, "", bgcolor = color.new(color.black, 0))
    r += 1
    table.cell(dash, 0, r, f_t("Mode", "Modo"), text_color = color.gray)
    table.cell(dash, 1, r, matchMode, text_color = color.white)
    r += 1
    table.cell(dash, 0, r, f_t("Match", "Correspondência"), text_color = color.gray)
    table.cell(dash, 1, r, str.tostring(trueCount) + "/" + str.tostring(enabledCount), text_color = color.white)
    r += 1
    table.cell(dash, 0, r, f_t("Live Priority", "Prioridade Atual"), text_color = color.gray)
    table.cell(dash, 1, r, priorityLabel, text_color = priorityLabel == "HIGH" ? color.lime : priorityLabel == "MEDIUM" ? color.yellow : priorityLabel == "LOW" ? color.orange : color.gray)
    r += 1
    table.cell(dash, 0, r, f_t("Cooldown", "Cooldown"), text_color = color.gray)
    table.cell(dash, 1, r, cooldownRemaining > 0 ? str.tostring(cooldownRemaining) + f_t(" bars", " barras") : f_t("Ready", "Pronto"), text_color = cooldownRemaining > 0 ? color.gray : color.lime)
    r += 1
    table.cell(dash, 0, r, f_t("Last Alert", "Último Alerta"), text_color = color.gray)
    table.cell(dash, 1, r, na(lastAlertBar) ? "—" : str.tostring(bar_index - lastAlertBar) + f_t(" bars ago", " barras atrás"), text_color = color.white)
    r += 1
    table.cell(dash, 0, r, f_t("CRITERIA", "CRITÉRIOS"), text_color = color.white, bgcolor = color.new(color.gray, 70), text_size = size.tiny)
    table.cell(dash, 1, r, "", bgcolor = color.new(color.gray, 70))
    r += 1
    if enabledCount > 0
        for i = 0 to enabledCount - 1
            table.cell(dash, 0, r, array.get(names, i), text_color = color.gray, text_size = size.tiny)
            table.cell(dash, 1, r, array.get(states, i) ? "✅" : "❌", text_color = array.get(states, i) ? color.lime : color.gray, text_size = size.tiny)
            r += 1
    table.cell(dash, 0, r, f_t("⚠ Not a trade signal", "⚠ Não é sinal de compra/venda"), text_color = color.orange, text_size = size.tiny)
    table.cell(dash, 1, r, f_t("You decide.", "A decisão é sua."), text_color = color.orange, text_size = size.tiny)

if showHistPanel and barstate.islast
    var table hist = table.new(f_pos(histPos), 1, maxHistory + 1, border_width = 1, bgcolor = color.new(color.black, 15))
    table.cell(hist, 0, 0, f_t("ALERT HISTORY", "HISTÓRICO DE ALERTAS"), text_color = color.white, bgcolor = color.new(color.black, 0), text_size = size.tiny)
    hn = array.size(historyLog)
    if hn == 0
        table.cell(hist, 0, 1, f_t("No alerts yet", "Ainda sem alertas"), text_color = color.gray, text_size = size.tiny)
    else
        for i = 0 to hn - 1
            table.cell(hist, 0, i + 1, array.get(historyLog, i), text_color = color.white, text_size = size.tiny)

bgcolor(useSessionFilter and sessionOK ? color.new(color.blue, 92) : na, title = "Required Session")

plot(ratio * 100, "Match Ratio %", display = display.data_window)
plot(priorityLabel == "HIGH" ? 3 : priorityLabel == "MEDIUM" ? 2 : priorityLabel == "LOW" ? 1 : 0, "Priority Level", display = display.data_window)
